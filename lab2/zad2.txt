SELECT first_name || ' ' || last_name AS imie_nazwisko,
       salary,
       phone_number
FROM   employees
WHERE  SUBSTR(last_name, 3, 1) = 'e'
AND    LOWER(first_name) LIKE '%' || LOWER('&czesc_imienia') || '%'
ORDER  BY 1 DESC, 2 ASC;


SELECT first_name,
       last_name,
       miesiace,
       CASE
           WHEN miesiace < 150              THEN salary * 0.10
           WHEN miesiace BETWEEN 150 AND 200 THEN salary * 0.20
           ELSE                                   salary * 0.30
       END AS wysokosc_dodatku
FROM  (SELECT first_name, last_name, salary,
              ROUND(MONTHS_BETWEEN(SYSDATE, hire_date)) AS miesiace
       FROM   employees)
ORDER BY miesiace;

WHERE  l.city = 'Toronto';


SELECT j.first_name || ' ' || j.last_name AS jennifer,
       c.first_name || ' ' || c.last_name AS wspolpracownik
FROM   employees j
LEFT   JOIN employees c
       ON  c.department_id = j.department_id
       AND c.employee_id  <> j.employee_id
WHERE  j.first_name = 'Jennifer'
ORDER  BY jennifer, wspolpracownik;

SELECT e.first_name, e.last_name, e.job_id,
       d.department_name, e.salary, g.grade_level
FROM   employees   e
LEFT   JOIN departments d ON d.department_id = e.department_id
JOIN   job_grades  g ON e.salary BETWEEN g.lowest_sal AND g.highest_sal;

SELECT d.department_name,
       COUNT(e.employee_id)      AS liczba_pracownikow,
       ROUND(AVG(e.salary), 2)   AS srednia_pensja
FROM   departments d
LEFT   JOIN employees e ON e.department_id = d.department_id
GROUP  BY d.department_name
ORDER  BY liczba_pracownikow DESC;

SELECT j.job_title, ROUND(AVG(e.salary), 2) AS srednie_zarobki
FROM   employees e
JOIN   jobs      j ON j.job_id = e.job_id
GROUP  BY j.job_title
ORDER  BY srednie_zarobki DESC
FETCH FIRST 5 ROWS ONLY;


SELECT r.region_name,
       COUNT(DISTINCT c.country_id) AS liczba_krajow,
       COUNT(e.employee_id)         AS liczba_pracownikow
FROM   regions r
LEFT   JOIN countries   c ON c.region_id     = r.region_id
LEFT   JOIN locations   l ON l.country_id    = c.country_id
LEFT   JOIN departments d ON d.location_id   = l.location_id
LEFT   JOIN employees   e ON e.department_id = d.department_id
GROUP  BY r.region_name;


SELECT e.first_name, e.last_name, e.salary AS zarobki_pracownika,
       m.salary AS zarobki_menedzera
FROM   employees e
JOIN   employees m ON m.employee_id = e.manager_id
WHERE  e.salary > m.salary;

ORDER  BY miesiac;
